// ════════════════════════════════════════════════════════════════════
// EXTRAÇÃO DE VARIÁVEIS AMBIENTAIS POR POLÍGONO - MapBiomas + ERA5
// Variáveis: Área, Altitude, Temperatura, Umidade, Água,
//            Distância Urbana, Distância ao Polígono Mais Próximo
// ════════════════════════════════════════════════════════════════════

// Carregar o shapefile de interesse (filtrando apenas polígonos)
var shapefile = ee.FeatureCollection("users/luizaklippel/MAPBIOMAS/cnuc_2025_08")
  .map(function(feature) {
    return feature.set('geomType', feature.geometry().type());
  })
  .filter(ee.Filter.inList('geomType', ['Polygon', 'MultiPolygon']));

var anos = ee.List.sequence(1985, 2024);

var mapbiomas = ee.Image(
  'projects/mapbiomas-public/assets/brazil/lulc/collection10/mapbiomas_brazil_collection10_coverage_v2'
);

// ════════════════════════════════════════════════════
// 1. ÁREA DE CADA POLÍGONO (hectares - estático)
// ════════════════════════════════════════════════════
var shapefileArea = shapefile.map(function(feature) {
  var areaHa = feature.geometry().area(1)
    .divide(10000); // m² → hectares
  return feature.set('Area_ha', areaHa).select(['Area_ha']);
});

Export.table.toDrive({
  collection: shapefileArea,
  description: 'Area_por_Poligono',
  fileFormat: 'CSV'
});

// ════════════════════════════════════════════════════
// 2. DISTÂNCIA AO POLÍGONO MAIS PRÓXIMO (km - estático)
// ════════════════════════════════════════════════════
var shapefileComCentroide = shapefile.map(function(feature) {
  var centroide = feature.geometry().centroid(1);
  return feature.set({
    'lon': centroide.coordinates().get(0),
    'lat': centroide.coordinates().get(1),
    'id_proprio': feature.id()
  });
});

var joinResult = ee.Join.saveBest({
  matchKey: 'vizinho',
  measureKey: 'distancia_m'
}).apply({
  primary: shapefileComCentroide,
  secondary: shapefileComCentroide,
  condition: ee.Filter.and(
    ee.Filter.withinDistance({
      distance: 500000, // 500km
      leftField: '.geo',
      rightField: '.geo',
      maxError: 1
    }),
    ee.Filter.notEquals({
      leftField: 'id_proprio',
      rightField: 'id_proprio'
    })
  )
});

var distanciaVizinho = joinResult.map(function(feature) {
  var distKm = ee.Number(feature.get('distancia_m')).divide(1000);
  return feature.set('DistViz_km', distKm).select(['DistViz_km']);
});

Export.table.toDrive({
  collection: distanciaVizinho,
  description: 'Distancia_Vizinho_por_Poligono',
  fileFormat: 'CSV'
});
// ════════════════════════════════════════════════════
// 3. ALTITUDE (SRTM - estático, não varia por ano)
// ════════════════════════════════════════════════════
var srtm = ee.Image('USGS/SRTMGL1_003').select('elevation');

var altitudeStats = srtm.reduceRegions({
  collection: shapefile,
  reducer: ee.Reducer.mean(),
  scale: 30
});

var shapefileAltitude = altitudeStats.map(function(f) {
  return f.set('Alt_media', f.get('mean')).select(['Alt_media']);
});

Export.table.toDrive({
  collection: shapefileAltitude,
  description: 'Altitude_Media_por_Poligono',
  fileFormat: 'CSV'
});

// ════════════════════════════════════════════════════
// 4. TEMPERATURA MÉDIA ANUAL (ERA5-Land, °C)
// ════════════════════════════════════════════════════
var imagemTemp = ee.Image(
  anos.iterate(function(ano, imgAcumulada) {
    var anoInt = ee.Number(ano).int();
    var bandaNome = ee.String('T_').cat(ee.String(anoInt));

    var tempAnual = ee.ImageCollection('ECMWF/ERA5_LAND/MONTHLY_AGGR')
      .filter(ee.Filter.calendarRange(anoInt, anoInt, 'year'))
      .select('temperature_2m')
      .mean()
      .subtract(273.15)
      .rename(bandaNome);

    return ee.Image(imgAcumulada).addBands(tempAnual);
  }, ee.Image([]))
);

var tempStats = imagemTemp.reduceRegions({
  collection: shapefile,
  reducer: ee.Reducer.mean(),
  scale: 11132
});

Export.table.toDrive({
  collection: tempStats,
  description: 'Temperatura_Media_por_Poligono',
  fileFormat: 'CSV'
});

// ════════════════════════════════════════════════════
// 5. UMIDADE RELATIVA MÉDIA ANUAL (ERA5-Land, %)
// ════════════════════════════════════════════════════
var imagemUmid = ee.Image(
  anos.iterate(function(ano, imgAcumulada) {
    var anoInt = ee.Number(ano).int();
    var bandaNome = ee.String('U_').cat(ee.String(anoInt));

    var colecao = ee.ImageCollection('ECMWF/ERA5_LAND/MONTHLY_AGGR')
      .filter(ee.Filter.calendarRange(anoInt, anoInt, 'year'));

    var T  = colecao.select('temperature_2m').mean().subtract(273.15);
    var Td = colecao.select('dewpoint_temperature_2m').mean().subtract(273.15);

    var rh = (Td.multiply(17.625).divide(Td.add(243.04)).exp())
      .divide(T.multiply(17.625).divide(T.add(243.04)).exp())
      .multiply(100)
      .rename(bandaNome);

    return ee.Image(imgAcumulada).addBands(rh);
  }, ee.Image([]))
);

var umidStats = imagemUmid.reduceRegions({
  collection: shapefile,
  reducer: ee.Reducer.mean(),
  scale: 11132
});

Export.table.toDrive({
  collection: umidStats,
  description: 'Umidade_Media_por_Poligono',
  fileFormat: 'CSV'
});

// ════════════════════════════════════════════════════
// 6. ÁREA DE CORPOS D'ÁGUA (MapBiomas, hectares)
// Classes: 33 = rios/lagos, 31 = aquicultura, 34 = glaciares
// ════════════════════════════════════════════════════
var imagemAgua = ee.Image(
  anos.iterate(function(ano, imgAcumulada) {
    var anoInt = ee.Number(ano).int();
    var bandaNome = ee.String('W_').cat(ee.String(anoInt));

    var classificacao = mapbiomas
      .select(ee.String('classification_').cat(ee.String(anoInt)));

    var agua = classificacao.eq(33)
      .or(classificacao.eq(31))
      .or(classificacao.eq(34));

    var areaAgua = agua
      .multiply(ee.Image.pixelArea())
      .divide(10000) // m² → hectares
      .rename(bandaNome);

    return ee.Image(imgAcumulada).addBands(areaAgua);
  }, ee.Image([]))
);

var aguaStats = imagemAgua.reduceRegions({
  collection: shapefile,
  reducer: ee.Reducer.sum(),
  scale: 30
});

Export.table.toDrive({
  collection: aguaStats,
  description: 'Area_Agua_por_Poligono',
  fileFormat: 'CSV'
});

// ════════════════════════════════════════════════════
// 7. DISTÂNCIA AO CENTRO URBANO MAIS PRÓXIMO (km)
// Classe 24 = Área Urbanizada no MapBiomas
// ════════════════════════════════════════════════════
var imagemDistancia = ee.Image(
  anos.iterate(function(ano, imgAcumulada) {
    var anoInt = ee.Number(ano).int();
    var bandaNome = ee.String('D_').cat(ee.String(anoInt));

    var classificacao = mapbiomas
      .select(ee.String('classification_').cat(ee.String(anoInt)));

    var urbano = classificacao.eq(24);

    var distancia = urbano
      .fastDistanceTransform(2048)
      .sqrt()
      .multiply(30)    // pixels → metros
      .divide(1000)    // metros → km
      .rename(bandaNome);

    return ee.Image(imgAcumulada).addBands(distancia);
  }, ee.Image([]))
);

var distanciaStats = imagemDistancia.reduceRegions({
  collection: shapefile,
  reducer: ee.Reducer.min(),
  scale: 30
});

Export.table.toDrive({
  collection: distanciaStats,
  description: 'Distancia_Urbano_por_Poligono',
  fileFormat: 'CSV'
});
